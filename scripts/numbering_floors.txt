# ── SCOPE (this is the load-bearing decision, not the numbers) ──────────────
#
# Two lanes, measured 2026-09-04 and re-cut 2026-10-29 when the stem
# classifier (numbering_gate.classify_stems, ported from riir-train's
# issue-556 gate) retired the decidability objection for `.benchmarks/`:
#
# SERIAL (raw-counted: `max_duplicate_numbers`, the above-highwater check, and
# the historical walk) — directories whose leading NNN_ is an ALLOCATOR-ISSUED
# SERIAL:
#   .plans/ .issues/ .research/ .proposals/
#
# FAMILY (CLASSIFIED, never raw-counted) — there the leading number is the
# OWNING plan/issue and a family per owner is the intended convention
# (010_* x5 are all Issue 010's "Report the Floor" benches; 294_ict_g1 / g2 /
# g3 / g10 / goat_gates / promotion are all Plan 294's gates), so a raw
# duplicate count would produce ~55 findings, all false. That objection was
# decidability; the classifier retired it — owner identity IS decidable from
# the stems — so `.benchmarks/` JOINS the scan under the classifier:
#
#   COLLISION        walls at 0 and is NEVER pinnable (the issue-556 law: a
#                    pinned COLLISION row is a work item, never an excuse; the
#                    fix is the loser's renumber + every citation moved in the
#                    same commit — measured 2026-10-29 on riir-ai 809/872/1008
#                    and riir-infer 884, all four renumbered exactly this way)
#   RENAME-RESIDUE   ratcheted at `max_rename_residue`
#   HAND-READ        ratcheted at `max_hand_read` — a new owner family whose
#                    stems carry no owner tokens lands here; naming the owner
#                    in the stem (`294_plan294_g1`) classifies it
#                    FAMILY-EXPLICIT instead, which is silent — the escape
#                    hatch is the convention, not an exemption
#   FAMILY-EXPLICIT  the intended convention; silent by construction
#
# The raw ratchet and the historical walk still EXCLUDE family dirs: a family
# per owner is growth, and ratcheting the convention itself would red on every
# legitimate bench. The family dir DOES get a floor, above-highwater and
# malformed-allocator checks like any other scanned directory (the 2026-09-04
# "700 while 701 existed" incident was this directory's allocator).
#
#   .docs/        — no flat NNN_ files at all (nested by topic: 01_orientation,
#                  02_inference, ...) and no .highwater. Nothing to check.
#
# ── FLOORS (not ceilings — read the reason) ─────────────────────────────
#
# Every verdict in this gate is a CEILING (0 duplicates, max <= highwater), and
# a ceiling cannot fail once the instrument goes blind: one regex regression
# takes the population to zero and every ceiling passes, indistinguishable from
# a clean repo. That is the failure mode this file's floors exist to catch, and
# it is the same lesson as scripts/percentile_floors.txt's min_sites_scanned.
#
# Set BELOW the measured count with headroom, because files are legitimately
# REMOVED here per the noise-reduction rule — a floor at the measured value
# would red on every correct removal.

min_files.issues = 0           # measured 0 (2026-09-05) — and 0 is the SUCCESS
                               # state here, not a blind walk: every issue was
                               # resolved and removed per the noise-reduction
                               # rule (723/724/725/726/727 all closed). The
                               # floor was 1 and RED on develop for exactly that
                               # reason; a gate that fails when the work is
                               # finished is measuring the wrong thing.
                               # Blindness is still caught, twice over: the
                               # three floors around this one breach if the walk
                               # returns nothing, and `.issues/.highwater`
                               # parses (727), which proves the dir is reached.
                               # Raise this again if issues become long-lived"

# ── the FAMILY lane (classified, never raw-counted — see SCOPE above) ──────
# Comma-list of family directories. Each MUST carry a min_files floor (the
# gate refuses an undeclared family population) and the two ratchet scalars
# below MUST be present (a silent default would wall a ratchet at 0 or green
# a class over its blindness).
family_dirs = .benchmarks

                               # two spellings is a defect in any class; a
                               # wall, and it re-pins only downward-never

                               # that grows the population (a new owner
                               # family) re-pins it; growth without the re-pin
                               # reds. Naming the owner in the stem moves a
                               # group to FAMILY-EXPLICIT (silent) instead.

# ── CEILINGS ────────────────────────────────────────────────────────────────
#
# A duplicate among TRACKED files is the defect: every cross-reference in this
# workspace is by number, so `Plan N` resolving to two documents means a reader
# following a citation cannot tell which. Untracked duplicates are reported in
# their own class and do NOT fail — a colleague's in-flight file is not a repo
# defect (the same distinction scripts/skill_repo_set_gate.py already draws for
# its untracked population) — but they fail the moment they are committed.
max_duplicate_numbers = 0

# A .highwater file that is PRESENT but not a bare integer is worse than a
# missing one: `scan()` used to swallow the ValueError and return None, which is
# also what an ABSENT allocator returns — so the ceiling above passed over a
# corrupted allocator and a clean directory identically. Measured 2026-09-05
# across all 16 contract repos: FIVE such files, every one of them the same
# cause — `echo -n <N> > .highwater` under a shell whose builtin `echo` does not
# implement `-n`, so the flag lands in the file (`-n 872`, `-n 374`, ...).
# katgpt-rs has none, which is exactly why the blindness survived here.
# Cross-repo verdict: scripts/numbering_drift_sweep.py.
max_malformed_highwater = 0

# A .highwater BELOW its directory's max is the precursor to a collision, not a
# cosmetic mismatch: `value + 1` is already taken, so the next allocation is
# guaranteed to collide. Found in TWO directories on 2026-09-04 (.plans 585 vs
# 586, .benchmarks 700 vs 701) — i.e. the loaded state is the normal state
# unless something checks.
max_above_highwater = 0
