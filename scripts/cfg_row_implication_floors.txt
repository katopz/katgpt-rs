# Pins for scripts/cfg_row_implication_gate.py (Issue 513).
#
# The question: does a target's `required-features` row, resolved WITH the
# package's default features, satisfy the target source's own leading
# `#![cfg(...)]`? If not, the row builds and compiles the target to NOTHING —
# `running 0 tests / ok. 0 passed`, exit 0 — and every other instrument in this
# repo reports it as protected.
#
# Measured 2026-09-06 (katgpt-rs): 623 rows, 393 with a leading #![cfg],
# 392 SATISFIED, 1 NO-FEATURE-REQ, 0 EMPTY-AT-ROW, 0 UNRESOLVED.

# WALL, not a ratchet. A row that compiles its own target to nothing is never
# legitimate — unlike an #[ignore], which has legitimate uses.
max_empty_at_row = 0

# Also a wall, at this repo's measured 0 — but read it as the narrow claim it
# is. UNRESOLVED means a predicate the report declines to rule on: `any(feature
# = ...)` (which cargo's AND-only required-features cannot express at all), or
# a feature under `not(...)` (where the implication INVERTS). Siblings
# legitimately have them; this repo happens not to. If a legitimate one lands
# here, raise this pin and say which row and why — do not delete the key.
#
# ── 2026-09-09 (Issue 741): 0 -> 2, and both rows are named, because a bare
#    count would let a third one in silently. The two:
#
#      katgpt-rs:test:kimi_k3_g4_alloc_free
#        #![cfg(all(feature = "kimi_k3_loader",
#                   any(debug_assertions, feature = "alloc_tracking")))]
#      katgpt-rs:test:bench_284_clr_goat_g4
#        #![cfg(feature = "clr")]
#        #![cfg(any(debug_assertions, feature = "alloc_tracking"))]
#
#    Both are G4 zero-alloc gates and both are UNRESOLVED for a reason this
#    report cannot fix: the predicate contains `debug_assertions`, so whether
#    the row compiles the target to nothing is a function of the PROFILE, which
#    a static pass over manifests + sources does not have. Declining to rule is
#    the correct behaviour — the alternative is a confident answer that is
#    wrong in one of the two profiles.
#
#    What the report cannot decide, measurement can, and it was measured in
#    every configuration rather than pinned on an argument:
#
#      row features only (clr / kimi_k3_loader), DEV      -> BUILDS, 1 passed
#      row features only, RELEASE                         -> empty binary, `ok. 0 passed`
#      row features + alloc_tracking, RELEASE             -> BUILDS, 1 passed
#
#    So the row is NOT the "row builds and compiles to nothing" defect this
#    gate hunts (`max_empty_at_row`, still a WALL at 0): at the row's own
#    feature set the target really does compile and really does assert, in the
#    profile the row is exercised in. The release green zero is the separate
#    PROFILE axis, and it is owned by a pin that CAN see it —
#    `cfg_gated_floors.txt`'s `max_profile_debug_only`, which reads the
#    UNFIXABLE half and is a WALL at 0. Issue 741 is what moved these two out
#    of the unfixable column by giving the allocator module a feature
#    (`alloc_tracking`) instead of gating it on the profile.
#
#    Do NOT raise this pin for a third `any(...)` row without the same
#    three-configuration measurement. The pin is 2 because two rows were
#    measured, not because `any()` is generically acceptable.
max_unresolved = 2

# FLOORS, with slack below the measured value. Both ceilings above are green
# over whatever the instrument can NAME, so a manifest-walk or source-scanner
# regression takes the population toward 0 and both pass — indistinguishable
# from a clean repo. Not hypothetical: an early cut of the scanner reported two
# riir-train targets as "source unreadable" because they use cargo's DIRECTORY
# form (`tests/<name>/main.rs`) and silently skipped them.
#
# Slack is deliberate. A row legitimately leaves when a target is deleted, and
# a ratcheting floor would red the next correct removal.
min_rows_scanned = 560
min_with_cfg = 359
